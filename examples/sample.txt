/* 1. Funzione Ricorsiva: testa chiamate annidate e stack frame */
fn fib(n) {
    if (n <= 0) {
        return 0;
    }
    if (n == 1) {
        return 1;
    }
    // Ricorsione multipla: valuta due rami e li somma!
    return fib(n - 1) + fib(n - 2);
}

/* 2. Funzione a 4 parametri: Geometria analitica (Pitagora) */
fn distanza(x1, y1, x2, y2) {
    dx = x2 - x1;
    dy = y2 - y1;
    // Formula: sqrt(dx^2 + dy^2)
    return (dx ^ 2 + dy ^ 2) ^ 0.5;
}

/* 3. Simulatore Fisico: Moto Parabolico con gravita' */
fn simula_proiettile(v0) {
    t = 0;
    dt = 0.5;
    g = 9.81;
    max_quota = 0;

    print("--- Inizio traiettoria ---");
    while (1 == 1) {
        // Formula: y(t) = v0*t - 0.5*g*t^2
        quota = (v0 * t) - (0.5 * g * (t ^ 2));

        if (quota > max_quota) {
            max_quota = quota;
        }

        // Se tocca il suolo dopo essere partito, ferma la simulazione con BREAK
        if (t > 0 && quota <= 0) {
            print("Impatto al suolo!");
            break;
        }

        print("Tempo: ", t, "s | Quota: ", quota);
        t = t + dt;
    }
    return max_quota;
}
